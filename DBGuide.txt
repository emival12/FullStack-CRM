# Creazione nuovo DB
CREATE DATABASE MarkotexDB;

# Selezione di un DB con cui lavorare
USE MarkotexDB;

# Creazione Tabella object_definition
#   La tabella contiene la definizione di tutti le tabelle del database con alcune informazioni chiave: 
#   la categoria di appartenenza, l'rdinamento, se è una tabella di sistema o se possiede molteplici record type
CREATE TABLE object_definition (
    object_name VARCHAR(255) NOT NULL,
    category VARCHAR(50) NOT NULL,
    sort_order SMALLINT,
    is_system_object BOOL DEFAULT 0,
    is_single_record_type BOOL DEFAULT 1,
    PRIMARY KEY (object_name),
    INDEX idx_category(category)
);

# Creazione Tabella field_definition
#   La tabella contiene la definizione di tutti i campi di tutte le tabelle del database più info chiave del campo.
#   Questa tabella è fondamentale per le lookup, in modo da conoscere a quale oggetto puntano
#   Il campo field_type avrà dei valori attesi di questo tipo: text, int, float, rollup_summary, lookup, picklist
CREATE TABLE field_definition (
    object_name VARCHAR(255),
    field_name VARCHAR(255),
    field_type VARCHAR(20) NOT NULL,
    length INT,
    numeric_precision TINYINT,
    numeric_scale TINYINT,
    reference_object VARCHAR(255),
    is_editable BOOL DEFAULT 1,
    is_required BOOL DEFAULT 0,
    is_active BOOL DEFAULT 1,
    PRIMARY KEY (object_name, field_name),
    FOREIGN KEY (object_name) REFERENCES object_definition(object_name) ON DELETE CASCADE,
    INDEX idx_reference_object(reference_object)
);

# Creazione Tabella record_type_definition
#   La tabella contiene la definizione dei record type degli oggetti
CREATE TABLE record_type_definition (
    object_name VARCHAR(255),
    record_type_name VARCHAR(255),
    is_active BOOL DEFAULT 1,
    PRIMARY KEY (object_name, record_type_name),
    FOREIGN KEY (object_name) REFERENCES object_definition(object_name) ON DELETE CASCADE
);

# Creazione Tabella record_type_field_definition
#   La tabella descrive i campi aperti sul record type, va combinata alla tabella field_definition per ottenere le info complete
#   Viene duplicato is_required e is_active perchè un campo potrebbe non essere attivo per un determinato recordType 
#   (però la sua lunghezza e il tipo è coerente tra i record type)
CREATE TABLE record_type_field_definition (
    object_name VARCHAR(255),
    record_type_name VARCHAR(255),
    field_name VARCHAR(255),
    is_editable BOOL DEFAULT 1,
    is_required BOOL DEFAULT 0,
    is_active BOOL DEFAULT 1,
    PRIMARY KEY (object_name, record_type_name, field_name),
    FOREIGN KEY (object_name, field_name) REFERENCES field_definition(object_name, field_name) ON DELETE CASCADE,
    FOREIGN KEY (object_name, record_type_name) REFERENCES record_type_definition(object_name, record_type_name) ON DELETE CASCADE
);



# Creazione Tabella aggregation_function, conterrà valori come: SUM, MIN, MAX, etc...
CREATE TABLE aggregation_function (
    name VARCHAR(15) PRIMARY KEY
);

# Creazione Tabella rollup_definition
#   La tabella descrive la struttura di un campo rollup, la funzione da applicare e gli eventuali filtri
CREATE TABLE rollup_definition (
    master_object_name VARCHAR(255),
    master_field_name VARCHAR(255),
    detail_object_name VARCHAR(255) NOT NULL,
    detail_field_name VARCHAR(255) NOT NULL,
    aggregation_function VARCHAR(15),
    filter_condition VARCHAR(1000),
    PRIMARY KEY (master_object_name, master_field_name, aggregation_function),
    FOREIGN KEY (master_object_name, master_field_name) REFERENCES field_definition(object_name, field_name) ON DELETE CASCADE,
    FOREIGN KEY (aggregation_function) REFERENCES aggregation_function(name)
);
